// Costume probe (2026-10-15): six costumes redrawn to the zodiac onesies' standard: each changes his whole silhouette, has one
// strong colour, and carries one big thing a child knows it by. Same layers as costumeLayer() (back, body, head, tail), plus
// pawL / pawR: drawn inside each paw's group, after the paw, so a held prop (hose, wok, staff, rabbit) and gloves move with it.
import { crescent } from '../cel';

export interface CostumeLayer { back: string; body: string; head: string; tail?: string; pawL?: string; pawR?: string; hidesEars: boolean }

/** His paws' outlines (parts.ts G.pawL / G.pawR), for gloves and for a paw closing over a held thing. */
const PAW_L_D = 'M122 262 C120 276 154 278 154 266 C154 258 124 254 122 262Z';
const PAW_R_D = 'M198 262 C200 276 166 278 166 266 C166 258 196 254 198 262Z';
const glove = (d: string, fill: string, toes: string) => `<path d="${d}" fill="${fill}" ${s2}/><path d="${toes}" fill="none" stroke="${celShade(fill)}" stroke-width="1.6" stroke-linecap="round"/>`;
const gloveL = (fill: string) => glove(PAW_L_D, fill, 'M132 264 L132 270 M142 264 L142 270');

/**
 * A cel piece (spec §7): its flat fill, its one hard shadow, then the line. The shadow is the piece itself in its shadow tone
 * with the lit fill on top shrunk toward (ax, ay), a point near its upper left: what is left is a rim on the side away from the
 * light. Explicit paths only (no clip-path); exact for any piece that is star-shaped about (ax, ay).
 */
const cel = (d: string, fill: string, ax: number, ay: number, k = 0.9, sh = celShade(fill)) =>
  `<path d="${d}" fill="${sh}"/><path d="${d}" fill="${fill}" transform="translate(${ax} ${ay}) scale(${k}) translate(${-ax} ${-ay})"/><path d="${d}" fill="none" ${S}/>`;
/** A soft white highlight stroke on the lit side. */
const hi = (d: string, w = 4, o = 0.5) => `<path d="${d}" fill="none" stroke="#fff" stroke-opacity="${o}" stroke-width="${w}" stroke-linecap="round"/>`;
/** A thick stroked band with an ink edge (a ribbon, a hose, a staff). */
const rope = (d: string, color: string, w: number) => ropeTail(d, color, w);

/**
 * COSTUME RULE (parent, 2026-10-15): headgear reaches both edges of the head; the crown of the head stays visible. A helmet's
 * brim, a headband, a circlet or a fall of hair runs right across his head to its outline on both sides (x ≈ 54–266 at the brow,
 * the cheek tufts at y 100–146), so no grey head shows beyond either end; but a band or circlet never grows into a cap over his
 * crown. Ears: as each design wants (out over a band, tucked under a helmet or hair).
 */
